Write a Solidity smart contract named BankAccount to implement a simple bank account system using blockchain technology



#Code

 // SPDX-License-Identifier: MIT
 // License information for this program

 pragma solidity ^0.8.0;
 // Use Solidity version 0.8.0 or above

 contract BankAccount {
 // Create a smart contract named BankAccount

     mapping(address => uint) balance;
     // Store each user's bank balance using their wallet address

     function deposit() public payable {
     // Function to deposit ETH into the account

         balance[msg.sender] += msg.value;
         // Add deposited ETH to the user's balance
     }

     function withdraw(uint amount) public {
     // Function to withdraw the given amount of ETH

         require(balance[msg.sender] >= amount, "Insufficient balance");
         // Check whether the user has enough balance

         balance[msg.sender] -= amount;
         // Subtract withdrawn amount from the balance

         payable(msg.sender).transfer(amount);
         // Send the withdrawn ETH to the user
     }

     function viewBalance() public view returns(uint) {
     // Function to check the user's balance

         return balance[msg.sender];
         // Return the current user's balance
     }
 }
 // End of smart contract

Lab-4
Write a Solidity program using Structures & Arrays to store Student Data. Deploy & observe gas usage.


#Code
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.0;                // Solidity version

contract StudentData {               // Create contract

    struct Student {                         // Store one student's details
        uint id;                     	                // Example: 1
        string name;                         // Example: Rahul
        uint marks;                          // Example: 85
    }

    Student[] students;               // Array to store many students

    function add(uint id, string memory name, uint marks) 
   public {
        students.push(Student(id, name, marks));       // Example: Add 1, Rahul, 85
    }

    function get(uint i) 
    public view returns(uint, string memory, uint) {
        Student memory s = students[i];                           // Example: Read student at index 0
        return (s.id, s.name, s.marks);                              	// Output: 1, Rahul, 85
    }
}
